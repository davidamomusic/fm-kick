# FM-KICK 0.2.0: p-locks of any parameter, the kick of a step too; REC clears them
wait 700
expect locks 0
tap EDIT
key w4 down
turn K2 5
wait 30
key w4 up
expect lk4 1
expect lval4.1 36
expect snd1 31
# the type of the kick, locked on a step (KICK page, K1)
key w8 down
turn K1 2
key w8 up
expect lval8.0 5
# PRESETS with the step held: another kick for that step
key w4 down
spin PRESET 2
key w4 up
expect kit4 2
expect lk4 2
# the step plays its locks
tap PLAY
wait 600
expect tune_now 36
wait 400
expect type_now 5
tap PLAY
# REC with the step held: its p-locks gone
key w4 down
tap REC
key w4 up
expect lk4 0
expect trig4 1
expect rec 0
# OCT- + GLO: the kick's parameters and the trigs at random, with p-locks; never the tempo
press OCT-
tap GLO
release OCT-
expect mode 3
wait 50
expect locks >0
expect bpm 1280
expect rec 0
# OCT+ + GLO: the same and the LFOs
expect mtgt0 0
press OCT+
tap GLO
release OCT+
expect bpm 1280
tap SAVE
wait 50
reboot

expect locks >0
tap SEQ
